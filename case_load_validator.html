These tests define the behaviour. They never call a live model:
MERIDIAN_MODEL_MODE=template forces the deterministic fallback draft.
"""
import json
import os
import subprocess
import sys

import pytest

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
import app  # noqa: E402

REQUIRED_FIELDS = ["draft_reply", "evidence_refs", "missing_information", "review_status"]

REFUND_CLAIMS = ["refund has been approved", "refund is approved", "approved a refund",
                 "we have refunded", "your refund will", "refund approved"]
RESOLVED_CLAIMS = ["issue is resolved", "has been resolved", "is now resolved", "marked as resolved"]
WINDOW_CLAIMS = ["inside the return window", "outside the return window", "within the return window",
                 "within two days", "within the two-day", "eligible for a return"]


@pytest.fixture(autouse=True)
def template_mode(monkeypatch):
    monkeypatch.setenv("MERIDIAN_MODEL_MODE", "template")


def lower_text(result):
    """Only the generated wording is checked. The customer's own words may legitimately
    contain phrases such as 'is resolved' (that is the request being examined)."""
    generated = [result.get("draft_reply", ""), result.get("case_summary", ""),
                 result.get("human_action_required", "")]
    return " ".join(str(v) for v in generated).lower()


def test_output_has_required_fields():
    result = app.review_case("C1")
    for field in REQUIRED_FIELDS:
        assert field in result, f"missing field {field}"
    assert isinstance(result["draft_reply"], str)
    assert isinstance(result["evidence_refs"], list)
    assert isinstance(result["missing_information"], list)
    assert result["review_status"] in {"READY_FOR_HUMAN_REVIEW", "NEEDS_INFORMATION", "BLOCKED"}


def test_c1_complete_case_is_ready_for_human_review():
    result = app.review_case("C1")
    assert result["review_status"] == "READY_FOR_HUMAN_REVIEW"
    assert result["missing_information"] == []
    assert "P1" in result["evidence_refs"]
    assert "P4" in result["evidence_refs"]
    text = lower_text(result)
    assert not any(c in text for c in REFUND_CLAIMS), "C1 must not approve a refund"
    assert not any(c in text for c in RESOLVED_CLAIMS), "C1 issue is open; must not say resolved"


def test_c2_missing_delivery_date_needs_information():
    result = app.review_case("C2")
    assert result["review_status"] == "NEEDS_INFORMATION"
    assert result["missing_information"] == ["delivery_date"]
    assert "P2" in result["evidence_refs"]
    text = lower_text(result)
    assert "delivery date" in text, "the draft must ask for the delivery date"
    assert not any(c in text for c in WINDOW_CLAIMS), "no return-window conclusion without a date"


def test_c2_gate_runs_before_the_model(monkeypatch):
    calls = []

    def fake_model(context):
        calls.append(context)
        return "SHOULD NOT BE USED"

    monkeypatch.setattr(app, "draft_with_model", fake_model)
    result = app.review_case("C2")
    assert calls == [], "the missing-field gate must stop the model call"
    assert result["review_status"] == "NEEDS_INFORMATION"


def test_c3_open_issue_is_not_called_resolved():
    result = app.review_case("C3")
    assert result["review_status"] == "READY_FOR_HUMAN_REVIEW"
    assert "P3" in result["evidence_refs"]
    text = lower_text(result)
    assert not any(c in text for c in RESOLVED_CLAIMS), "P3: open issue must not be called resolved"
    assert "open" in text or "under review" in text or "still" in text


def test_cli_prints_json_for_each_case():
    env = dict(os.environ, MERIDIAN_MODEL_MODE="template")
    for case_id in ["C1", "C2", "C3"]:

"""Meridian Retail support-draft prototype (Class 2 baseline).

Reads one prepared case from cases.json and four rules from policy.md,
then returns a structured draft for a human reviewer.

It cannot send a message, approve a refund or change a record (P4).

Model use:
  MERIDIAN_MODEL_MODE = auto (default) | template | live
  auto  -> live model if ANTHROPIC_API_KEY is set, otherwise the template draft
  MERIDIAN_MODEL = model id for live mode (default claude-sonnet-5)
"""
import argparse
import json
import os
import re
import sys

REQUIRED_FIELDS = ["delivery_date"]  # fixed gate, checked in code before any model call
ALLOWED_STATUS = {"READY_FOR_HUMAN_REVIEW", "NEEDS_INFORMATION", "BLOCKED"}

REFUND_CLAIMS = ["refund has been approved", "refund is approved", "approved a refund",
                 "we have refunded", "your refund will", "refund approved"]
RESOLVED_CLAIMS = ["issue is resolved", "has been resolved", "is now resolved", "marked as resolved"]
WINDOW_CLAIMS = ["inside the return window", "outside the return window", "within the return window",
                 "within two days", "within the two-day", "eligible for a return"]


# ---------------------------------------------------------------- inputs
def load_cases(path="cases.json"):
    with open(path, encoding="utf-8") as f:
        return {c["case_id"]: c for c in json.load(f)}


def load_policy(path="policy.md"):
    text = open(path, encoding="utf-8").read()
    version = re.search(r"policy_version:\s*(\S+)", text)
    rules = dict(re.findall(r"^(P\d)\.\s+(.*)$", text, flags=re.M))
    return {"policy_version": version.group(1) if version else "UNKNOWN", "rules": rules}


# ---------------------------------------------------------------- fixed checks (normal code)
def check_required_fields(case):
    return [f for f in REQUIRED_FIELDS if case.get(f) in (None, "", [])]


def boundary_violations(draft, case, missing):
    text = draft.lower()
    problems = [c for c in REFUND_CLAIMS if c in text]
    if case.get("issue_status") == "open":
        problems += [c for c in RESOLVED_CLAIMS if c in text]
    if "delivery_date" in missing:
        problems += [c for c in WINDOW_CLAIMS if c in text]
    return problems


# ---------------------------------------------------------------- the model step
SYSTEM_PROMPT = (
    "You draft customer-support replies for a human reviewer at Meridian Retail. "
    "Use ONLY the supplied case fields and policy rules. Do not approve or reject a return, "
    "do not mention a refund decision, do not say an open issue is resolved, do not claim the request "
    "is inside or outside any return window, and do not invent facts. "
    "Write 2-4 plain sentences. Return only the reply text."
)


def template_draft(context):
    case = context["case"]
    product, order_id = case["product"], case["order_id"]
    if "resolved" in case["customer_question"].lower() and case["issue_status"] == "open":
        return (f"Thank you for checking in about order {order_id}. Our records show this issue is still open: "
                f"{case['issue_details'].lower()}. A member of our team is reviewing it and will confirm the next "
                f"step once there is an update.")
    return (f"Thank you for letting us know about the {product.lower()} on order {order_id} "
            f"({case['issue_details'].lower()}). The delivery date we have on file is {case['delivery_date']}. "
            f"A member of our team will review whether the report can be referred for a return under our policy "
            f"and will confirm the next step. No refund decision has been made yet.")


def live_draft(context):
    import urllib.request
    payload = {
        "model": os.environ.get("MERIDIAN_MODEL", "claude-sonnet-5"),
        "max_tokens": 300,
        "system": SYSTEM_PROMPT,
        "messages": [{"role": "user", "content": json.dumps(context, indent=2)}],
    }
    req = urllib.request.Request(
        "https://api.anthropic.com/v1/messages",
        data=json.dumps(payload).encode("utf-8"),
        headers={"content-type": "application/json",
                 "x-api-key": os.environ["ANTHROPIC_API_KEY"],
                 "anthropic-version": "2023-06-01"},
    )
    with urllib.request.urlopen(req, timeout=40) as resp:
        data = json.loads(resp.read().decode("utf-8"))
    return "".join(b.get("text", "") for b in data.get("content", []) if b.get("type") == "text").strip()


def draft_with_model(context):
    """Return proposed wording. Live model if configured, otherwise the labelled template."""
    mode = os.environ.get("MERIDIAN_MODEL_MODE", "auto").lower()
    use_live = mode == "live" or (mode == "auto" and os.environ.get("ANTHROPIC_API_KEY"))
    if use_live:
        try:
            text = live_draft(context)
            print(f"[model] live {os.environ.get('MERIDIAN_MODEL', 'claude-sonnet-5')}", file=sys.stderr)
            return text
        except Exception as exc:  # network, key or format problem -> fall back, say so
            print(f"[model] live call failed ({exc}); using template draft", file=sys.stderr)
    else:
        print("[model] template draft (no live model configured)", file=sys.stderr)
    return template_draft(context)


# ---------------------------------------------------------------- the review
def review_case(case_id, cases=None, policy=None):
    cases = cases or load_cases()
    policy = policy or load_policy()
    case = cases[case_id]
    missing = check_required_fields(case)

    # Gate 1: missing required fact -> ask for it, do not reason about the return window, no model call
    if missing:
        print("[model] not called: required field missing", file=sys.stderr)
        return {
            "draft_reply": (f"Thank you for your message about the {case['product'].lower()} on order "
                            f"{case['order_id']}. To check whether it can be returned we need the delivery date "
                            f"for this order. Could you confirm when it was delivered? A member of our team will "
                            f"then review the request."),
            "evidence_refs": ["P2", "P4"],
            "missing_information": missing,
            "review_status": "NEEDS_INFORMATION",
        }

    evidence_refs = ["P1", "P4"] if "return" in case["customer_question"].lower() else ["P4"]
    if case["issue_status"] == "open":
        evidence_refs.append("P3")
    evidence_refs = sorted(set(evidence_refs))

    context = {"task": "Draft a reply for human review. Do not decide the return or the refund.",
               "policy_version": policy["policy_version"],
               "rules": {k: policy["rules"][k] for k in evidence_refs if k in policy["rules"]},
               "case": case}
    draft = draft_with_model(context)

    # Gate 2: check the wording against the boundaries after the model call
    problems = boundary_violations(draft, case, missing)
    if problems:
        print(f"[guard] draft violated boundaries {problems}; replaced with template", file=sys.stderr)
        draft = template_draft(context)

    return {
        "draft_reply": draft,
        "evidence_refs": evidence_refs,
        "missing_information": [],
        "review_status": "READY_FOR_HUMAN_REVIEW",
    }


def main(argv=None):
    parser = argparse.ArgumentParser(description="Meridian support-draft prototype")
    parser.add_argument("--case", required=True, help="case_id from cases.json, e.g. C1")
    args = parser.parse_args(argv)
    result = review_case(args.case)
    assert result["review_status"] in ALLOWED_STATUS
    print(json.dumps(result, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())

Ask

C2: which policy decision cannot be made from the evidence, and what should happen next?

Ask

C3: can the draft confirm the issue is resolved? Which rule controls the answer?

No. The customer asks for confirmation; asking does not prove resolution. The Issue Record says open; P3 forbids saying resolved. Keep both pieces of information visible and keep the case in human review.

Learn
  • Missing information is a condition the case is in, with its own next step, not an empty cell to guess around.
  • C3 is not a mysterious system conflict. It is a requested statement versus a supplied authoritative status. Do not teach software to pick the more plausible one.

"""Meridian Retail support-draft prototype (Class 4: evidence-linked Case Review).

Keeps the four Class 2 fields (draft_reply, evidence_refs, missing_information,
review_status) with the same meaning and type, and adds:
case_id, case_summary, known_facts, evidence_links, conflicting_information,
human_action_required. Source IDs come only from evidence_map.json.

It cannot send a message, approve a refund or change a record (P4).

Model use:
  MERIDIAN_MODEL_MODE = auto (default) | template | live
  auto  -> live model if ANTHROPIC_API_KEY is set, otherwise the template draft
  MERIDIAN_MODEL = model id for live mode (default claude-sonnet-5)
"""
import argparse
import json
import os
import re
import sys

REQUIRED_FIELDS = ["delivery_date"]  # fixed gate, checked in code before any model call
ALLOWED_STATUS = {"READY_FOR_HUMAN_REVIEW", "NEEDS_INFORMATION", "BLOCKED"}
FACT_FIELDS = ["case_id", "order_id", "product", "delivery_date", "issue_status", "issue_details"]

REFUND_CLAIMS = ["refund has been approved", "refund is approved", "approved a refund",
                 "we have refunded", "your refund will", "refund approved"]
RESOLVED_CLAIMS = ["issue is resolved", "has been resolved", "is now resolved", "marked as resolved"]
WINDOW_CLAIMS = ["inside the return window", "outside the return window", "within the return window",
                 "within two days", "within the two-day", "eligible for a return"]


# ---------------------------------------------------------------- inputs
def load_cases(path="cases.json"):
    with open(path, encoding="utf-8") as f:
        return {c["case_id"]: c for c in json.load(f)}


def load_policy(path="policy.md"):
    text = open(path, encoding="utf-8").read()
    version = re.search(r"policy_version:\s*(\S+)", text)
    rules = dict(re.findall(r"^(P\d)\.\s+(.*)$", text, flags=re.M))
    return {"policy_version": version.group(1) if version else "UNKNOWN", "rules": rules}


def load_evidence_map(path="evidence_map.json"):
    with open(path, encoding="utf-8") as f:
        return json.load(f)


# ---------------------------------------------------------------- fixed checks (normal code)
def check_required_fields(case):
    return [f for f in REQUIRED_FIELDS if case.get(f) in (None, "", [])]


def boundary_violations(draft, case, missing):
    text = draft.lower()
    problems = [c for c in REFUND_CLAIMS if c in text]
    if case.get("issue_status") == "open":
        problems += [c for c in RESOLVED_CLAIMS if c in text]
    if "delivery_date" in missing:
        problems += [c for c in WINDOW_CLAIMS if c in text]
    return problems


def known_facts_for(case, sources):
    """Recorded values paired with a supplied source ID. Missing values are not facts."""
    facts = []
    for field in FACT_FIELDS:
        value = case.get(field)
        if value in (None, ""):
            continue
        if field in sources:  # never invent a source ID
            facts.append({"field": field, "value": value, "source_id": sources[field]})
    return facts


def evidence_links_for(case, sources):
    links = [{"claim": f"The customer asks: {case['customer_question']}", "source_id": sources["customer_question"]},
             {"claim": f"The issue record status is {case['issue_status']}.", "source_id": sources["issue_status"]}]
    if case.get("delivery_date"):
        links.append({"claim": f"The recorded delivery date is {case['delivery_date']}.",
                      "source_id": sources["delivery_date"]})
    return links


def conflicts_for(case, sources):
    """Keep a customer request that differs from the supplied record visible. Do not pick a side."""
    asks_resolution = "resolved" in case["customer_question"].lower()
    if asks_resolution and case.get("issue_status") == "open":
        return [{
            "topic": "issue_status",
            "customer_request": "Customer asks for confirmation that the issue is resolved.",
            "request_source_id": sources["customer_question"],
            "recorded_value": case["issue_status"],
            "record_source_id": sources["issue_status"],
            "rule": "P3",
        }]
    return []


# ---------------------------------------------------------------- the model step
SYSTEM_PROMPT = (
    "You draft customer-support replies for a human reviewer at Meridian Retail. "
    "Use ONLY the supplied case fields and policy rules. Do not approve or reject a return, "
    "do not mention a refund decision, do not say an open issue is resolved, do not claim the request "
    "is inside or outside any return window, and do not invent facts. "
    "Write 2-4 plain sentences. Return only the reply text."
)


def template_draft(context):
    case = context["case"]
    product, order_id = case["product"], case["order_id"]
    if "resolved" in case["customer_question"].lower() and case["issue_status"] == "open":
        return (f"Thank you for checking in about order {order_id}. Our records show this issue is still open: "
                f"{case['issue_details'].lower()}. A member of our team is reviewing it and will confirm the next "
                f"step once there is an update.")
    return (f"Thank you for letting us know about the {product.lower()} on order {order_id} "
            f"({case['issue_details'].lower()}). The delivery date we have on file is {case['delivery_date']}. "
            f"A member of our team will review whether the report can be referred for a return under our policy "
            f"and will confirm the next step. No refund decision has been made yet.")


def template_summary(case):
    product, order_id = case["product"].lower(), case["order_id"]
    if "resolved" in case["customer_question"].lower():
        return (f"The customer asks for confirmation of resolution for the {product} on order {order_id}; "
                f"the issue record says {case['issue_status']}.")
    return f"The customer asks whether the {product} on order {order_id} can be returned ({case['issue_details'].lower()})."


def live_draft(context):
    import urllib.request
    payload = {
        "model": os.environ.get("MERIDIAN_MODEL", "claude-sonnet-5"),
        "max_tokens": 300,
        "system": SYSTEM_PROMPT,
        "messages": [{"role": "user", "content": json.dumps(context, indent=2)}],
    }
    req = urllib.request.Request(
        "https://api.anthropic.com/v1/messages",
        data=json.dumps(payload).encode("utf-8"),
        headers={"content-type": "application/json",
                 "x-api-key": os.environ["ANTHROPIC_API_KEY"],
                 "anthropic-version": "2023-06-01"},
    )
    with urllib.request.urlopen(req, timeout=40) as resp:
        data = json.loads(resp.read().decode("utf-8"))
    return "".join(b.get("text", "") for b in data.get("content", []) if b.get("type") == "text").strip()


def draft_with_model(context):
    """Return proposed wording. Live model if configured, otherwise the labelled template."""
    mode = os.environ.get("MERIDIAN_MODEL_MODE", "auto").lower()
    use_live = mode == "live" or (mode == "auto" and os.environ.get("ANTHROPIC_API_KEY"))
    if use_live:
        try:
            text = live_draft(context)
            print(f"[model] live {os.environ.get('MERIDIAN_MODEL', 'claude-sonnet-5')}", file=sys.stderr)
            return text
        except Exception as exc:
            print(f"[model] live call failed ({exc}); using template draft", file=sys.stderr)
    else:
        print("[model] template draft (no live model configured)", file=sys.stderr)
    return template_draft(context)


# ---------------------------------------------------------------- the review
def review_case(case_id, cases=None, policy=None, evidence_map=None):
    cases = cases or load_cases()
    policy = policy or load_policy()
    evidence_map = evidence_map or load_evidence_map()
    case = cases[case_id]
    sources = evidence_map["cases"][case_id]
    missing = check_required_fields(case)
    known_facts = known_facts_for(case, sources)
    links = evidence_links_for(case, sources)
    conflicts = conflicts_for(case, sources)

    # Gate 1: missing required fact -> ask for it, no return-window reasoning, no model call
    if missing:
        print("[model] not called: required field missing", file=sys.stderr)
        return {
            "case_id": case_id,
            "case_summary": template_summary(case),
            "known_facts": known_facts,
            "evidence_refs": ["P2", "P4"],
            "evidence_links": links,
            "missing_information": missing,
            "conflicting_information": conflicts,
            "draft_reply": (f"Thank you for your message about the {case['product'].lower()} on order "
                            f"{case['order_id']}. To check whether it can be returned we need the delivery date "
                            f"for this order. Could you confirm when it was delivered? A member of our team will "
                            f"then review the request."),
            "review_status": "NEEDS_INFORMATION",
            "human_action_required": "Confirm the delivery date from the agreed source, then return the case for review.",
        }

    evidence_refs = ["P1", "P4"] if "return" in case["customer_question"].lower() else ["P4"]
    if case["issue_status"] == "open":
        evidence_refs.append("P3")
    evidence_refs = sorted(set(evidence_refs))

    context = {"task": "Draft a reply for human review. Do not decide the return or the refund.",
               "policy_version": policy["policy_version"],
               "rules": {k: policy["rules"][k] for k in evidence_refs if k in policy["rules"]},
               "case": case,
               "known_facts": known_facts,
               "conflicting_information": conflicts}
    draft = draft_with_model(context)

    # Gate 2: check the wording against the boundaries after the model call
    problems = boundary_violations(draft, case, missing)
    if problems:
        print(f"[guard] draft violated boundaries {problems}; replaced with template", file=sys.stderr)
        draft = template_draft(context)

    if conflicts:
        action = ("Confirm the current issue status with the owner of the Issue Record and decide the reply. "
                  "Do not confirm resolution while the record says open.")
    else:
        action = ("Review whether the reported damage can be referred for return review under P1, "
                  "then record the decision in the Human Review Log.")

    return {
        "case_id": case_id,
        "case_summary": template_summary(case),
        "known_facts": known_facts,
        "evidence_refs": evidence_refs,
        "evidence_links": links,
        "missing_information": [],
        "conflicting_information": conflicts,
        "draft_reply": draft,
        "review_status": "READY_FOR_HUMAN_REVIEW",
        "human_action_required": action,
    }


def main(argv=None):
    parser = argparse.ArgumentParser(description="Meridian support-draft prototype")
    parser.add_argument("--case", required=True, help="case_id from cases.json, e.g. C1")
    args = parser.parse_args(argv)
    result = review_case(args.case)
    assert result["review_status"] in ALLOWED_STATUS
    print(json.dumps(result, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())

# ------------------------------------------------------------------ small helpers
def read_csv(path):
    with open(path, newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        return list(reader.fieldnames or []), [dict(r) for r in reader]


def write_csv(path, columns, rows):
    os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=columns, extrasaction="ignore")
        w.writeheader()
        for r in rows:
            w.writerow({c: r.get(c, "") for c in columns})


def parse_iso(ts):
    try:
        return datetime.fromisoformat(ts.replace("Z", "+00:00"))
    except Exception:
        return None


def valid_date(text):
    try:
        datetime.strptime(text, "%Y-%m-%d")
        return True
    except Exception:
        return False


def file_hash(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()[:12]


def table(rows, columns, max_width=42):
    """Plain-text table for the terminal."""
    if not rows:
        return "  (no rows)"
    def cut(v):
        v = "" if v is None else str(v)
        return v if len(v) <= max_width else v[: max_width - 1] + "…"
    widths = {c: max(len(c), *(len(cut(r.get(c, ""))) for r in rows)) for c in columns}
    line = "  " + "  ".join(c.ljust(widths[c]) for c in columns)
    sep = "  " + "  ".join("-" * widths[c] for c in columns)
    body = ["  " + "  ".join(cut(r.get(c, "")).ljust(widths[c]) for c in columns) for r in rows]
    return "\n".join([line, sep] + body)


def banner(title):
    print("\n" + "=" * 78 + f"\n  {title}\n" + "=" * 78)


# ------------------------------------------------------------------ the pipeline
class Pipeline:
    def __init__(self, data_dir=DATA_DIR, out_dir=OUT_DIR, run_time=RUN_TIME, quiet=False):
        self.data_dir, self.out_dir, self.run_time, self.quiet = data_dir, out_dir, run_time, quiet
        self.sources = {}      # name -> (columns, rows)
        self.raw = {}          # name -> (columns, rows with lineage)
        self.results = []      # every validation result, PASS included
        self.file_flags = []   # LOAD_WARNING / PAUSE_LOAD
        self.routes = {}       # (table, source_row_number) -> route
        self.rules = []

    def say(self, *a):
        if not self.quiet:
            print(*a)

    # -- stage 1: load ------------------------------------------------------------
    def load(self, override=None):
        banner("STAGE 1  LOAD  (ingestion: files -> working area)")
        rows_out = []
        for name in OPERATIONAL_FILES + ["validation_rules"]:
            path = os.path.join(self.data_dir, f"{name}.csv")
            if not os.path.exists(path):
                self.file_flags.append({"file": f"{name}.csv", "route": "PAUSE_LOAD", "rule_id": "VR-007",
                                        "received_value": "file missing",
                                        "reason": "Every required source file and required column must exist."})
                continue
            cols, rows = read_csv(path)
            if override and name in override:          # prepared failure (schema drift)
                cols, rows = override[name]
            self.sources[name] = (cols, rows)
            extract = rows[0].get("extract_generated_at", "") if rows else ""
            age = ""
            if extract and parse_iso(extract):
                age = f"{(self.run_time - parse_iso(extract)).total_seconds() / 3600:.0f} h"
            rows_out.append({"file": f"{name}.csv", "rows": len(rows), "extract_generated_at": extract or "(rule book)",
                             "age_at_run_time": age, "content_hash": file_hash(path)})
        self.rules = [r for r in self.sources.get("validation_rules", ([], []))[1] if r.get("enabled", "true") == "true"]
        self.say(table(rows_out, ["file", "rows", "extract_generated_at", "age_at_run_time", "content_hash"]))
        self.say(f"\n  Meridian's clock for this run: {self.run_time.strftime('%Y-%m-%d %H:%M UTC')}")
        self.say("  Files received ≠ data ready. Nothing has been checked yet.")
        return rows_out

    # -- stage 2: raw ----------------------------------------------------------------
    def land_raw(self):
        banner("STAGE 2  RAW  (exact copy + lineage: the crime-scene photograph)")
        loaded_at = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
        recon = []
        for name in OPERATIONAL_FILES:
            if name not in self.sources:
                continue
            cols, rows = self.sources[name]
            raw_rows = []
            for i, r in enumerate(rows, start=1):
                rr = dict(r)
                rr.update({"source_file": f"{name}.csv", "source_row_number": str(i), "loaded_at": loaded_at})
                raw_rows.append(rr)
            self.raw[name] = (cols + LINEAGE, raw_rows)
            write_csv(os.path.join(self.out_dir, f"raw_{name}.csv"), cols + LINEAGE, raw_rows)

        self.say("  source_file, source_row_number, loaded_at  ->  written to out/raw_*.csv")
        return recon

                             "record_key": row.get(KEY_COLUMN.get(tbl, ""), ""), "field": field,
                             "received_value": value, "route": route, "reason": reason})

        # VR-007 schema: required files and columns
        for name in OPERATIONAL_FILES:
            if name not in self.sources:
                continue
            cols = self.sources[name][0]
            missing = [c for c in REQUIRED_COLUMNS[name] if c not in cols]
            if missing:
                self.file_flags.append({"file": f"{name}.csv", "route": "PAUSE_LOAD", "rule_id": "VR-007",
                                        "received_value": f"missing column(s): {', '.join(missing)}",
                                        "reason": "Every required source file and required column must exist."})
        if any(f["route"] == "PAUSE_LOAD" for f in self.file_flags):
            self.say("  PAUSE_LOAD  the whole delivery cannot be trusted. Nothing is routed, nothing is published.")
            self.say(table(self.file_flags, ["file", "rule_id", "route", "received_value", "reason"], 60))
            return False

        for name in OPERATIONAL_FILES:

            extract = parse_iso(rows[0].get("extract_generated_at", "")) if rows else None
            age_h = (self.run_time - extract).total_seconds() / 3600 if extract else None
            if age_h is None or age_h > FRESHNESS_HOURS:
                self.file_flags.append({"file": f"{name}.csv", "route": "LOAD_WARNING", "rule_id": "VR-008",
                                        "received_value": f"extract_generated_at {rows[0].get('extract_generated_at', '?')} "
                                                          f"({age_h:.0f} h old)" if age_h else "no extract time",
                                        "reason": f"File older than {FRESHNESS_HOURS} h at Meridian's run time. "
                                                  "Rows still flow; be careful what you claim from them."})

        orders = {r["order_id"]: r for r in self.raw["orders"][1]}
        issues = {r["case_id"]: r for r in self.raw["issue_records"][1]}

        bad_orders = set()
        for o in self.raw["orders"][1]:
            d = o.get("delivery_date", "")
            if d and not valid_date(d):
                bad_orders.add(o["order_id"])
                self._result("VR-005", "orders", o, "delivery_date", d, "REJECT",
                             "Not a valid YYYY-MM-DD date; blocks the linked case bundle.")
            else:
                self._result("VR-005", "orders", o, "delivery_date", d, "PASS", "")

        id_counts = {}
        for c in cases:
            id_counts[c.get("case_id", "")] = id_counts.get(c.get("case_id", ""), 0) + 1
        for c in cases:
            cid, oid, ctype, status = c.get("case_id", ""), c.get("order_id", ""), c.get("case_type", ""), c.get("review_status", "")
            # VR-001 present
            if not cid:
                self._result("VR-001", "support_cases", c, "case_id", "(blank)", "REJECT",
                             "case_id missing: cannot be traced or connected. Never invent an ID.")
            else:
                self._result("VR-001", "support_cases", c, "case_id", cid, "PASS", "")
            # VR-002 unique
            if cid and id_counts[cid] > 1:
                self._result("VR-002", "support_cases", c, "case_id", cid, "REJECT",
                             f"case_id appears {id_counts[cid]} times; both copies fail. Fix at the source.")
            elif cid:
                self._result("VR-002", "support_cases", c, "case_id", cid, "PASS", "")
            # VR-003 conditional required
            if ctype in NEEDS_ORDER and not oid:
                self._result("VR-003", "support_cases", c, "order_id", "(blank)", "REJECT",
                             f"order_id required when case_type is {ctype}.")
            else:
                self._result("VR-003", "support_cases", c, "order_id", oid or "(blank, not required)", "PASS", "")
            # VR-004 allowed values
            if status not in ALLOWED_STATUS:
                self._result("VR-004", "support_cases", c, "review_status", status, "REJECT",
                             "Value outside the contract (PENDING, APPROVED, REJECTED). Ask Meridian: should the contract change?")
            else:
                self._result("VR-004", "support_cases", c, "review_status", status, "PASS", "")
            # VR-005 propagated: broken order blocks its case bundle
            if oid in bad_orders:
                self._result("VR-005", "support_cases", c, "order_id", oid, "REJECT",
                             f"Linked order {oid} failed the delivery_date rule; the case is only as good as its proof.")
            # VR-006 cross-source match
            o, iss = orders.get(oid), issues.get(cid)

                                 f"orders.csv={o['delivery_date']} | issue_records.csv={iss['reported_delivery_date']}",
                                 "HUMAN_REVIEW", "Two valid dates disagree. Source of truth not decided; a person must choose.")
                else:
