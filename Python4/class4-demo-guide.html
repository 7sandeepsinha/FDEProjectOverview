Story: Ops lead asks “Last night’s batch finished with no error. Did every case get checked, and can I trust the number?” Class 3’s program printed 4 of 7 cases need attention and we believed it. Today the Workbench learns to survive bad rows, broken rules, missing files and a failing model, and can prove every row is counted.

Three promises: (1) every source row is accounted for: in the report or in quarantine with a reason; (2) every failure has a controlled outcome: set aside, halt, or UNAVAILABLE; (3) a rerun never repeats business work.

Four kinds of failure → outcome: Malformed record → quarantine that row · Business-rule failure (dup C-3004) → duplicate policy · Missing/unreadable source → stop before work, name the file · Outside service fails → bounded retry, then UNAVAILABLE.

"""Defensive parsing: read a value only when its meaning is certain, otherwise refuse it clearly."""
 
import re
from datetime import date
from decimal import Decimal, InvalidOperation
 
from meridian_workbench.domain import OrderStatus
from meridian_workbench.errors import MalformedRecord
 
DATE_SHAPE = re.compile(r"^\d{4}-\d{2}-\d{2}$")
AMOUNT_SHAPE = re.compile(r"^\d+\.\d{2}$")  # money: digits, a dot, exactly two decimal places
UNITS_SHAPE = re.compile(r"^\d+$")  # a whole, non-negative count
 
 
def require_text(raw: str | None, field: str) -> str:
    """A field must exist and not be blank. Surrounding spaces are removed."""
    if raw is None or not raw.strip():
        raise MalformedRecord(f"{field} is empty")
    return raw.strip()
 
 
def parse_date(raw: str | None, field: str = "promised_date") -> date:
    """Only YYYY-MM-DD is accepted. 07/03/2025 could mean 7 March or 3 July, so we refuse it."""
    text = require_text(raw, field)
    if not DATE_SHAPE.match(text):
        raise MalformedRecord(f"{field} {text!r} is not YYYY-MM-DD")
    try:
        return date.fromisoformat(text)
    except ValueError as error:
        raise MalformedRecord(f"{field} {text!r} is not a real calendar date") from error
 
 
def parse_units(raw: str | None, field: str = "units_on_hand") -> int:
    text = require_text(raw, field)
    if not UNITS_SHAPE.match(text):
        raise MalformedRecord(f"{field} {text!r} is not a whole number of zero or more")
    return int(text)
 
 
def parse_amount(raw: str | None, field: str = "unit_value") -> Decimal:
    """Money in our data contract: a dot, exactly two decimal places, no thousands separator."""
    text = require_text(raw, field)
    if not AMOUNT_SHAPE.match(text):
        raise MalformedRecord(
            f"{field} {text!r} must look like 19.99 (two decimal places, no commas)"
        )
    try:
        value = Decimal(text)
    except InvalidOperation as error:  # the shape check should catch this first; belt and braces
        raise MalformedRecord(f"{field} {text!r} is not a number") from error
    if not value.is_finite():
        raise MalformedRecord(f"{field} {text!r} is not a finite number")
    return value

    text = require_text(raw, field)
    try:
        return OrderStatus(text)
    except ValueError as error:

"""Batch-level checks: required files, duplicate policy, the quality gate and the books."""
 
from pathlib import Path
 
from meridian_workbench.domain import Ledger, RejectCategory, SupportCase
from meridian_workbench.errors import (
    MissingSource,
    QualityGateFailed,
    ReconciliationError,
    RuleFailure,
)
 
REQUIRED_SOURCES = ("cases.csv", "notes.json", "shipments.csv", "inventory.csv")
DUPLICATE_POLICIES = ("first_wins", "reject_batch")
 
 
def require_sources(data_dir: Path) -> None:
    """Check every required file before reading any of them. Name all the missing ones at once."""
    missing = [name for name in REQUIRED_SOURCES if not (data_dir / name).is_file()]
    if missing:
        raise MissingSource(f"required source(s) missing from {data_dir}: {', '.join(missing)}")
 
 
def apply_duplicate_policy(
    rows: list[tuple[int, SupportCase]], ledger: Ledger, policy: str
) -> list[tuple[int, SupportCase]]:
    """A case_id must appear once. first_wins keeps the first row; reject_batch stops the batch."""
    if policy not in DUPLICATE_POLICIES:
        raise ValueError(f"unknown duplicate policy {policy!r}; use one of {DUPLICATE_POLICIES}")
    first_row: dict[str, int] = {}
    kept: list[tuple[int, SupportCase]] = []
    for row_number, case in rows:
        if case.case_id in first_row:
            if policy == "reject_batch":
                raise RuleFailure(
                    f"case_id {case.case_id} appears on rows {first_row[case.case_id]} "
                    f"and {row_number}; duplicate_policy is reject_batch"
                )
            ledger.reject(
                row_number,
                RejectCategory.DUPLICATE,
                f"{case.case_id},{case.order_id}",
                f"case_id {case.case_id} already appears on row {first_row[case.case_id]}; "
                "first_wins keeps that row",
            )
            continue
        first_row[case.case_id] = row_number
        kept.append((row_number, case))
    ledger.accepted = len(kept)
    return kept
 
 
def check_quality(ledger: Ledger, max_share: float) -> None:
    """Halt when too many rows were quarantined, or when the file is empty."""
    if ledger.input_rows == 0:
        raise QualityGateFailed(
            f"{ledger.source} has no data rows; an empty file is a problem, not a quiet success"
        )
    quarantined = len(ledger.rejected)
    share = quarantined / ledger.input_rows
    if share > max_share:
        raise QualityGateFailed(
            f"{ledger.source}: {quarantined} of {ledger.input_rows} rows ({share:.0%}) were "
            f"quarantined; the limit is {max_share:.0%}"
        )
 
 
def reconcile(ledger: Ledger) -> None:
    """Every input row is accepted or quarantined. Anything else is a bug, and we say so."""
    quarantined = len(ledger.rejected)
    if ledger.accepted + quarantined != ledger.input_rows:
        difference = ledger.input_rows - ledger.accepted - quarantined
        raise ReconciliationError(
            f"{ledger.source}: {ledger.input_rows} rows in, {ledger.accepted} accepted and "
            f"{quarantined} quarantined; difference of {difference} (not zero)"
        )

"""Bounded retry with exponential backoff and full jitter. Only transient failures are retried."""
 
import random
import time
from collections.abc import Callable
from typing import TypeVar
 
from meridian_workbench.llm_client import LlmRateLimited, LlmServerError, LlmTimeout
 
T = TypeVar("T")
RETRYABLE: tuple[type[Exception], ...] = (LlmTimeout, LlmRateLimited, LlmServerError)
 
 
def backoff_delay(attempt: int, base: float, cap: float, rng: random.Random) -> float:
    """A random wait from 0 up to the current ceiling. The ceiling doubles, up to the cap."""
    ceiling = min(cap, base * 2 ** (attempt - 1))
    return rng.uniform(0, ceiling)
 
 
def call_with_retry(
    operation: Callable[[], T],
    *,
    max_attempts: int,
    base: float,
    cap: float,
    rng: random.Random,
    sleep: Callable[[float], None] = time.sleep,
) -> T:
    if max_attempts < 1:
        raise ValueError("max_attempts must be at least 1")
    for attempt in range(1, max_attempts + 1):
        try:
            return operation()
        except RETRYABLE as error:
            if attempt == max_attempts:
                raise  # out of attempts: pass the original error up, with its traceback
            delay = backoff_delay(attempt, base, cap, rng)
            print(f"attempt {attempt} failed ({type(error).__name__}); waiting {delay:.2f}s")
            sleep(delay)
    raise AssertionError("unreachable: the loop always returns or raises")
